
-- RyanX D1 Schema - Paste di Cloudflare D1 Console
PRAGMA foreign_keys=ON;

-- USERS
CREATE TABLE IF NOT EXISTS users (
  id TEXT PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  email TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  xpoin INTEGER DEFAULT 1000,
  role TEXT DEFAULT 'user',
  avatar TEXT DEFAULT 'default',
  headgear TEXT DEFAULT 'none',
  outfit TEXT DEFAULT 'basic',
  other TEXT DEFAULT 'none',
  last_snake_game INTEGER DEFAULT 0,
  created_at INTEGER
);
-- CONFIG GLOBAL - anti inflasi
CREATE TABLE IF NOT EXISTS config (
  key TEXT PRIMARY KEY,
  value TEXT
);
INSERT OR IGNORE INTO config(key,value) VALUES

-- ITEMS MALL
CREATE TABLE IF NOT EXISTS shop_items (
  id TEXT PRIMARY KEY,
  category TEXT CHECK(category IN ('headgear','outfit','other')),
  name TEXT,
  price INTEGER,
  image TEXT,
  rarity TEXT,
  description TEXT
);

-- INVENTORY
CREATE TABLE IF NOT EXISTS inventory (
  user_id TEXT,
  item_id TEXT,
  PRIMARY KEY(user_id,item_id)
);

-- TRANSACTIONS LOG
CREATE TABLE IF NOT EXISTS xpoin_log (
  id TEXT PRIMARY KEY,
  user_id TEXT,
  type TEXT,
  amount INTEGER,
  tax INTEGER DEFAULT 0,
  note TEXT,
  created_at INTEGER
);

-- FORUM
CREATE TABLE IF NOT EXISTS forum_topics (
  id TEXT PRIMARY KEY,
  user_id TEXT,
  title TEXT,
  content TEXT,
  image_url TEXT,
  link_url TEXT,
  created_at INTEGER
);
CREATE TABLE IF NOT EXISTS forum_comments (
  id TEXT PRIMARY KEY,
  topic_id TEXT,
  user_id TEXT,
  content TEXT,
  created_at INTEGER
);

-- LOBBY STATE (play03)
CREATE TABLE IF NOT EXISTS lobby_rooms (
  room_id TEXT PRIMARY KEY,
  name TEXT,
  max_users INTEGER DEFAULT 15
);
INSERT OR IGNORE INTO lobby_rooms VALUES ('room1','Lobby 1',15),('room2','Lobby 2',15),('room3','Lobby 3',15);

CREATE TABLE IF NOT EXISTS lobby_presence (
  user_id TEXT PRIMARY KEY,
  room_id TEXT,
  x INTEGER,
  y INTEGER,
  status TEXT DEFAULT 'online',
  last_seen INTEGER
);

-- GAME HISTORY
CREATE TABLE IF NOT EXISTS game_history (
  id TEXT PRIMARY KEY,
  user_id TEXT,
  game TEXT,
  bet INTEGER,
  win INTEGER,
  result TEXT,
  created_at INTEGER
);

-- TRADING CANDLES
CREATE TABLE IF NOT EXISTS trade_candles (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  timeframe TEXT,
  open REAL,
  high REAL,
  low REAL,
  close REAL,
  timestamp INTEGER
);

-- DICE
CREATE TABLE IF NOT EXISTS dice_history (
  id TEXT PRIMARY KEY,
  user_id TEXT,
  bet_type TEXT,
  bet_value TEXT,
  dice_result TEXT,
  win INTEGER,
  created_at INTEGER
);
